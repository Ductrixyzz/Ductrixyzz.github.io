local function make(class: string, props: { [string]: any }, parent: Instance?): any
	local o = Instance.new(class)
	for k, v in pairs(props) do o[k] = v end
	o.Parent = parent

local function makeDraggable(handle: GuiObject, target: GuiObject)
	local dragging, dragStart, startPos = false, nil, nil
	trackConn(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = target.Position
		end
	end))
	trackConn(handle.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end))
	trackConn(UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end))
end

--[[ FPS BOOST ]]
local boostApplied = false

local function boostInstance(o: Instance)
	if o:IsA("ParticleEmitter") or o:IsA("Trail") or o:IsA("Beam") or o:IsA("Smoke") or o:IsA("Fire") or o:IsA("Sparkles") then
		o.Enabled = false
	elseif o:IsA("Decal") or o:IsA("Texture") then
		o.Transparency = 1
	elseif o:IsA("Explosion") then
		o.Visible = false
	end
end

local function applyBoostFPS()
	if boostApplied then return end
	boostApplied = true
	pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
	pcall(function() Lighting.GlobalShadows = false end)
	pcall(function() Lighting.FogEnd = 100000 end)
	pcall(function() Workspace.Terrain.Decoration = false end)
	for _, o in ipairs(Workspace:GetDescendants()) do
		pcall(boostInstance, o)
	end
	trackConn(Workspace.DescendantAdded:Connect(function(o) pcall(boostInstance, o) end))
end

local function invoke(...): any
	local remotes = ReplicatedStorage:FindFirstChild("Remotes")
	local commF = remotes and remotes:FindFirstChild("CommF_")
	if not commF then return nil end
	local args = table.pack(...)
	local ok, res = pcall(function()
		return commF:InvokeServer(table.unpack(args, 1, args.n))
	end)
	if ok then return res end
	return nil
end

local function isFruitTool(t: Instance): boolean
	return t:IsA("Tool") and string.find(t.Name, "-", 1, true) ~= nil
end

local function fruitBase(toolName: string): string
	local base = string.match(toolName, "^(.-)%-") or toolName
	return string.match(base, "^%s*(.-)%s*$") or toolName
end

local function isProtected(base: string): boolean
	if Config.ProtectMythicalFruits and MYTHICAL[base] then return true end
	if Config.ProtectLegendaryFruits and LEGENDARY[base] then return true end
	if not CHIP_RANK[base] then return true end
	return false
end

local function pickChip(): Tool?
	local best, bestRank = nil, math.huge
	for _, t in ipairs(LocalPlayer.Backpack:GetChildren()) do
		if isFruitTool(t) then
			local base = fruitBase(t.Name)
			local rank = CHIP_RANK[base]
			if rank and not isProtected(base) and rank < bestRank then
				best, bestRank = t, rank
			end
		end
	end
	return best
end

-- Cache inventory một lần mỗi tick thay vì gọi remote nhiều lần
local function snapshotInventory(): { [string]: boolean }
	local set = {}
	local function scan(t: any, depth: number)
		if depth > 4 or type(t) ~= "table" then return end
		for k, v in pairs(t) do
			if type(k) == "string" then set[k] = true end
			if type(v) == "string" then set[v] = true end
			if type(v) == "table" then scan(v, depth + 1) end
		end
	end
	scan(invoke(RM.Inventory), 0)
	for _, c in ipairs({ LocalPlayer.Backpack, getChar() }) do
		if c then
			for _, t in ipairs(c:GetChildren()) do set[t.Name] = true end
		end
	end
	return set
end

local function missingHaki(inv: { [string]: boolean }): { string }
	local miss = {}
	for _, color in ipairs(HAKI_COLORS) do
		if not inv[color] then table.insert(miss, color) end
	end
	return miss
end

local function equipWeapon(): Tool?
	local char = getChar()
	local hum = getHum()
	if not char or not hum then return nil end
	local held = char:FindFirstChildOfClass("Tool")
	if held and not isFruitTool(held) and not CHALICES[held.Name] then return held end
	for _, t in ipairs(LocalPlayer.Backpack:GetChildren()) do
		if t:IsA("Tool") and not isFruitTool(t) and not CHALICES[t.Name] then
			hum:EquipTool(t)
			return t
		end
	end
	return nil
end

	end
	return nil
end

--[[ FLY (bay từng bước theo tốc độ, không teleport) ]]
local function flyTo(target: Vector3, timeout: number): boolean
	local t0 = os.clock()
	local last = os.clock()
	while env.DarkwenRun == RUN and alive() and os.clock() - t0 < timeout do
		local hrp = getHRP()
		if not hrp then return false end
		local now = os.clock()
		local dt = math.min(now - last, 0.1)
		last = now
		local pos = hrp.Position
		local delta = target - pos
		local dist = delta.Magnitude
		if dist < 4 then return true end
		local speed = math.clamp(Config.FlySpeed, Config.FlySpeedMin, Config.FlySpeedMax)
		local newPos = pos + delta.Unit * math.min(dist, speed * dt)
		hrp.CFrame = CFrame.new(newPos)
		hrp.AssemblyLinearVelocity = Vector3.zero
		task.wait()
	end
	return false
end

local function safeEscape()
	local hrp = getHRP()
	if hrp then hrp.CFrame = hrp.CFrame + Vector3.new(0, 150, 0) end
	task.wait(3)
end

		if Config.BringMob and not S.busy and alive() then
			local now = os.clock()
			if now - S.lastSimSet > 3 then
				S.lastSimSet = now
				setSimulationRadius()
			end
			local hrp = getHRP()
			local folder = getEnemiesFolder()
			if hrp and folder then
				local count = 0
				for _, m in ipairs(folder:GetChildren()) do
					if count >= Config.BringMaxCount then break end
					local h = m:FindFirstChildOfClass("Humanoid")
					local r = m:FindFirstChild("HumanoidRootPart")
					if h and r and h.Health > 0 and not BRING_SKIP[m.Name] and (r.Position - hrp.Position).Magnitude <= Config.BringRadius then
						count += 1
						local x = ((count - 1) % 4 - 1.5) * 3
						r.CFrame = hrp.CFrame * CFrame.new(x, -2, -6)
						r.AssemblyLinearVelocity = Vector3.zero
						r.CanCollide = false
					end
				end
			end
		end
		task.wait(0.1)
	end
end

--[[ FAST ATTACK ]]
local function hasEnemyInRange(hrp: BasePart): boolean
	local folder = getEnemiesFolder()
	if not folder then return false end
	for _, m in ipairs(folder:GetChildren()) do
		local h = m:FindFirstChildOfClass("Humanoid")
		local r = m:FindFirstChild("HumanoidRootPart")
		if h and r and h.Health > 0 and (r.Position - hrp.Position).Magnitude <= Config.AttackRange then
			return true
		end
	end
	return false
end

local function fastAttackLoop()
	while env.DarkwenRun == RUN do
		if Config.FastAttack and not S.busy and alive() then
			local hrp = getHRP()
			if hrp and hasEnemyInRange(hrp) then
				local weapon = equipWeapon()
				if weapon then weapon:Activate() end
			end
		end
		task.wait(Config.FastAttackDelay)
	end
end

--[[ WEBHOOK (gửi 1 lần) ]]
local function sendWebhook(inv: { Mirror: boolean, Valk: boolean })
	if not Config.WebhookEnabled or Config.WebhookURL == "" or not req or env.DarkwenWebhookSent then return end
	env.DarkwenWebhookSent = true
	local name = LocalPlayer.Name
	local masked = string.sub(name, 1, 2) .. string.rep("*", math.max(#name - 4, 1)) .. string.sub(name, -2)
	local payload = HttpService:JSONEncode({
		embeds = {{
			title = BRAND .. " | Hoàn thành Mirror Fractal & Valkyrie Helm",
			color = 5763719,
			fields = {
				{ name = "Tài khoản", value = masked, inline = true },
				{ name = "Mảnh Gương", value = inv.Mirror and "Có" or "Chưa", inline = true },
				{ name = "Mũ Valkyrie", value = inv.Valk and "Có" or "Chưa", inline = true },
				{ name = "Fragments", value = tostring(getFragments()), inline = true },
				{ name = "Thời gian", value = os.date("%d/%m/%Y %H:%M:%S"), inline = false },
			},
		}},
	})
	task.spawn(function()
		pcall(req, {
			Url = Config.WebhookURL,
			Method = "POST",
			Headers = { ["Content-Type"] = "application/json" },
			Body = payload,
		})
	end)
end

--[[ SERVER HOP (mọi lần hop đều delay 5s) ]]
local function serverHop(reason: string): boolean
	setStatus(reason .. " | hop sau 5s")
	task.wait(HOP_DELAY)
	if not req then
		setStatus("Executor không hỗ trợ request")
		return false
	end
	local okFind, targetId = pcall(function()
		local res = req({
			Url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", game.PlaceId),
			Method = "GET",
		})
		local data = HttpService:JSONDecode(res.Body)
		for _, s in ipairs(data.data or {}) do
			if s.id ~= game.JobId and not P.visited[s.id] and s.playing < s.maxPlayers then
				return s.id
			end
		end
		return nil
	end)
	if not okFind or not targetId then
		setStatus("Không tìm thấy server phù hợp")
		return false
	end
	P.visited[targetId] = true
	P.hop += 1
	saveProgress()
	if Config.QueueOnTeleport and queueTeleport then
		pcall(queueTeleport, BOOT_SNIPPET)
	end
	local okTp = pcall(function()
		TeleportService:TeleportToPlaceInstance(game.PlaceId, targetId, LocalPlayer)
	end)
	return okTp
end

--[[ SAFE FRUIT HANDLER ]]
local function safeFruitHandler()
	if not Config.AutoStoreFruits then return end
	for _, t in ipairs(LocalPlayer.Backpack:GetChildren()) do
		if isFruitTool(t) then
			local base = fruitBase(t.Name)
			local mustStore = (Config.ProtectMythicalFruits and MYTHICAL[base])
				or (Config.ProtectLegendaryFruits and LEGENDARY[base])
			if mustStore then
				invoke(RM.StoreFruit, t.Name)
				task.wait(0.3)
			end
		end
	end
end

--[[ COMBAT ]]
-- Engage: bay tới sau lưng mục tiêu, việc đánh do fastAttackLoop đảm nhiệm
local function engage(mob: Model): boolean
	local hum = mob:FindFirstChildOfClass("Humanoid")
	local root = mob:FindFirstChild("HumanoidRootPart")
	if not hum or not root then return false end
	while env.DarkwenRun == RUN and mob.Parent and hum.Health > 0 and alive() do
		local hrp = getHRP()
		local myHum = getHum()
		if not hrp or not myHum then break end
		if Config.SafeModeWithChalice and hasChalice() and hpPercent(myHum) < 30 then
			setStatus("HP thấp khi cầm Cúp - né lên cao")
			safeEscape()
			return false
		end
		local target = (root.CFrame * CFrame.new(0, 0, 5)).Position
		flyTo(target, 1.5)
		task.wait(0.05)
	end
	return hum.Health <= 0
end

local function tryBoss(needMirror: boolean, needValk: boolean): boolean
	local targets = {}
	if needMirror then table.insert(targets, "Dough King") end
	if needValk then table.insert(targets, "rip_indra") end
	for _, kw in ipairs(targets) do
		local boss = findEnemy(kw)
		local hum = boss and boss:FindFirstChildOfClass("Humanoid")
		if boss and hum and hpPercent(hum) >= Config.MinBossHPPercent then
			setStatus("Đánh boss " .. kw)
			engage(boss)
			return true
		end
	end
	return false
end

local function summonWithChalice(chalice: Tool): boolean
	local hum = getHum()
	if not hum then return false end
	S.busy = true
	hum:EquipTool(chalice)
	task.wait(0.3)
	chalice:Activate()
	task.wait(3)
	S.busy = false
	return true
end

--[[ CHALICE ACTIONS ]]
local function farmCocoaAndSummonDK()
	if countItem("Conjured Cocoa") >= 10 and countItem("Sweet Chalice") == 0 then
		setStatus("Đổi Sweet Chalice")
		invoke(RM.SweetChalice, "Craft")
		task.wait(1)
	end
	if P.cake >= CAKE_KILLS_NEED then
		local chalice = findTool("Sweet Chalice")
		if chalice then
			setStatus("Triệu hồi Dough King")
			summonWithChalice(chalice)
		end
		return
	end
	local mob = findCakeMob()
	if not mob then
		setStatus("Chờ quái Cake Land")
		task.wait(2)
		return
	end
	setStatus(string.format("Farm Cake Land %d/%d", P.cake, CAKE_KILLS_NEED))
	if engage(mob) then
		P.cake += 1
		saveProgress()
	end
end

local function summonRipIndra(inv: { [string]: boolean }): boolean
	if #missingHaki(inv) > 0 then return false end
	local chalice = findTool("God's Chalice")
	if not chalice then return false end
	setStatus("Triệu hồi rip_indra")
	return summonWithChalice(chalice)
end

--[[ RAID / FRAGMENTS ]]
local function pickupFruitsOnMap()
	local hrp = getHRP()
	if not hrp then return end
	for _, obj in ipairs(Workspace:GetChildren()) do
		local handle = obj:IsA("Tool") and obj:FindFirstChild("Handle")
		if handle and isFruitTool(obj) then
			setStatus("Nhặt trái trên map")
			flyTo(handle.Position + Vector3.new(0, 3, 0), 5)
			if firetouchinterest then
				pcall(firetouchinterest, hrp, handle, 0)
				pcall(firetouchinterest, hrp, handle, 1)
			end
			task.wait(0.5)
			return
		end
	end
end

local function raidForFragments()
	local chip = pickChip()
	if not chip and Config.AutoRandomFruitForRaid then
		setStatus("Random trái để Raid")
		invoke(RM.RandomFruit, "Buy")
		task.wait(1)
		chip = pickChip()
	end
	if not chip then
		pickupFruitsOnMap()
		return
	end
	setStatus("Raid " .. Config.RaidType .. " lấy Fragments")
	invoke(RM.RaidSelect, "Start", Config.RaidType)
	local t0 = os.clock()
	while env.DarkwenRun == RUN and getFragments() < FRAG_NEED and os.clock() - t0 < RAID_TIMEOUT and alive() do
		local mob = findEnemy("")
		if mob then
			engage(mob)
		else
			task.wait(0.5)
		end
	end
end

local function prepareLegendaryHaki(inv: { [string]: boolean }): boolean
	if not Config.AutoHopLegendaryHaki then return false end
	local miss = missingHaki(inv)
	if #miss == 0 then return false end
	if getFragments() < FRAG_NEED then
		if Config.AutoRaidForFragments then
			raidForFragments()
			return true
		end
		return false
	end
	local cousin = Workspace:FindFirstChild("Barista Cousin", true)
	if not cousin then
		return serverHop("Chưa thấy Barista Cousin")
	end
	local root = cousin:FindFirstChild("HumanoidRootPart")
	if root then flyTo(root.Position + Vector3.new(0, 0, 4), 6) end
	task.wait(0.5)
	for _, color in ipairs(miss) do
		if getFragments() >= FRAG_NEED then
			setStatus("Mua Haki " .. color)
			invoke(RM.Cousin, "BuyHaki", color)
			task.wait(1)
		end
	end
	return true
end

local function collectChests()
	local hrp = getHRP()
	if not hrp then return end
	for _, obj in ipairs(Workspace:GetDescendants()) do
		if obj.Name == "Chest" and obj:IsA("BasePart") then
			setStatus("Mở rương tìm Cúp")
			flyTo(obj.Position + Vector3.new(0, 3, 0), 6)
			if firetouchinterest then
				pcall(firetouchinterest, hrp, obj, 0)
				pcall(firetouchinterest, hrp, obj, 1)
			end
			task.wait(0.5)
			return
		end
	end
end

--[[ FINISH ]]
local function finishRun(inv: { Mirror: boolean, Valk: boolean })
	S.done = true
	P.done = true
	if Config.StopAndIdleWhenDone then
		local hrp = getHRP()
		local spawnLoc = Workspace:FindFirstChildWhichIsA("SpawnLocation", true)
		if hrp then
			hrp.CFrame = (spawnLoc and spawnLoc.CFrame or CFrame.new(0, 100, 0)) + Vector3.new(0, 5, 0)
		end
		trackConn(LocalPlayer.Idled:Connect(function()
			pcall(function()
				VirtualUser:CaptureController()
				VirtualUser:ClickButton2(Vector2.new())
			end)
		end))
	end
	setStatus("Hoàn thành: Mảnh Gương " .. (inv.Mirror and "Có" or "Chưa") .. " | Mũ Valkyrie " .. (inv.Valk and "Có" or "Chưa"))
	sendWebhook(inv)
end

--[[ MAIN STATE MACHINE ]]
local function tick()
	if S.done then
		task.wait(2)
		return
	end
	if not alive() then
		setStatus("Chờ nhân vật hồi sinh")
		task.wait(1)
		return
	end

	safeFruitHandler()
	local snap = snapshotInventory()
	local inv = { Mirror = snap["Mirror Fractal"] == true, Valk = snap["Valkyrie Helm"] == true }
	S.inv = inv
	local needMirror = Config.AutoMirrorFractal and not inv.Mirror
	local needValk = Config.AutoValkyrieHelm and not inv.Valk

	if not needMirror and not needValk then
		finishRun(inv)
		return
	end

	-- Bước 2: đang cầm Cúp → khóa hop
	if hasChalice() then
		setStatus("Đang cầm Cúp - hop bị khóa")
		if tryBoss(needMirror, needValk) then return end
		if needMirror and Config.AutoFarmCocoaAndSummonDK then
			farmCocoaAndSummonDK()
			return
		end
		if needValk and Config.AutoSummonRipIndra and summonRipIndra(snap) then return end
		task.wait(1)
		return
	end

	-- Bước 3: Haki Legendary & Fragments
	if needValk and prepareLegendaryHaki(snap) then return end

	-- Bước 4: săn boss theo Hybrid
	if tryBoss(needMirror, needValk) then return end

	if Config.PreferHopBoss and P.hop < Config.MaxHopAttempts then
		return serverHop(string.format("Hop tìm boss (%d/%d)", P.hop, Config.MaxHopAttempts))
	end

	if Config.AutoFarmEliteForChalice then
		if needMirror and Config.AutoFarmCocoaAndSummonDK and countItem("Conjured Cocoa") < 10 then
			local cake = findCakeMob()
			if cake then
				setStatus("Farm Conjured Cocoa")
				engage(cake)
				return
			end
		end
		local elite = findEnemy("Elite Hunter")
		if elite then
			setStatus("Đánh Elite Hunter lấy God's Chalice")
			engage(elite)
			return
		end
		if Config.AutoCollectChest4H then collectChests() end
		return serverHop("Server không có Elite Hunter")
	end

	task.wait(1)
end

local function main()
	if not LocalPlayer.Character then
		LocalPlayer.CharacterAdded:Wait()
	end
	while env.DarkwenRun == RUN do
		local ok, err = pcall(tick)
		if not ok then
			setStatus("Lỗi: " .. tostring(err))
			task.wait(1)
		end
		task.wait(0.5)
	end
end
